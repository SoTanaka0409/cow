0162: 	else
0163: 	{
0164: 		mIsVacuumActive = false;
0165: 
0166: 		float recoverySpeed = VACUUM_RECOVER_PER_FRAME;

0171: 			auto timer = ServiceLocator::GetGameManager()->GetGameTimer();
0172: 			if (timer && timer->GetTime() <= 60)
0173: 			{
0174: 				recoverySpeed *= 2.0f;
0175: 			}
0176: 		}
0177: 
0178: 		mVacuumGauge += recoverySpeed;
0179: 		if (mVacuumGauge > VACUUM_GAUGE_MAX) mVacuumGauge = VACUUM_GAUGE_MAX;
0180: 	}
0181: 
0182: 	// フィーバータイム中はゲージに関係なく強制的に吸い込みが常時発動する
0183: 	if (Master::FeverFlag) mIsVacuumActive = true;
0184: }

0329: 	mvOldPosition = mvPosition; // 衝突前の現在座標を保存
0330: 	moveVec = VGet(0.0f, 0.0f, 0.0f);
0331:

0333: 	VECTOR leftMoveVector = VGet(0.0f, 0.0f, 0.0f); // カメラ基準の左方向ベクトル

0442: /*
0443:  * @brief 移動方向へ滑らかにキャラクターの向きを旋回させる処理（線形補間風）
0444:  */
0445: void Player3D::RotationByMove()
0446: {
0447: 	// 現在の角度と移動入力から求めた目標角度の差分を算出
0448: 	float subAngle = mfTargetAngle - mfAngle;
0449: 
0450: 	// 角度の不連続点（-180度?180度の境界線）を跨いだ場合の最短ルート補正
0451: 	if (subAngle < -DX_PI_F) subAngle += DX_TWO_PI_F;
0452: 	if (subAngle > DX_PI_F)  subAngle -= DX_TWO_PI_F;
0453: 
0454: 	// 毎フレーム一定の定数（ROTATE_SPEED）ずつ目標角度へと近づける
0455: 	if (subAngle > 0.0f)
0456: 	{
0457: 		subAngle -= ROTATE_SPEED;
0458: 		if (subAngle < 0.0f) subAngle = 0.0f;
0459: 	}
0460: 	else if (subAngle < 0.0f)
0461: 	{
0462: 		subAngle += ROTATE_SPEED;
0463: 		if (subAngle > 0.0f) subAngle = 0.0f;
0464: 	}
0465: 
0466: 	// 補正された角度を適用
0467: 	mfAngle = mfTargetAngle - subAngle;
0468: 
0469: 	// モデルの初期方向に合わせて180度（DX_PI_F）反転して反映
0470: 	mvRotation.y = mfAngle + DX_PI_F;
0471: 	mpModel->SetRotation(mvRotation);
0472: }

0503: /*
0504:  * @brief コライダー接触開始時のコールバック（牛やその他動物の吸引開始）
0505:  */
0506: void Player3D::OnEnter(Collider* collider, Collider* check)
0507: {
0508: 	// --- 「牛」のコライダーがプレイヤーの吸い込み範囲に入った場合 ---
0509: 	if (collider == mpCapsuleCollider && check->mpParentObject->GetTag() == Tag3D_Cow)
0510: 	{
0511: 		CowMove* cow = dynamic_cast<CowMove*>(check->mpParentObject);
0512: 		if (cow->GetCurrentState() != STATE_VACUUM)
0513: 		{
0514: 			cow->SetTargetPlayer(this);
0515: 			cow->IncreaseVacuumTimer();
0516: 			cow->ChangeStateToVacuum(); // 対象を吸引状態（上昇・回転）へ移行させる
0517: 		}
0518: 	}
0519: 	// --- 「その他動物」のコライダーがプレイヤーの吸い込み範囲に入った場合 ---
0520: 	if (collider == mpCapsuleCollider && check->mpParentObject->GetTag() == Tag3D_Animal)
0521: 	{
0522: 		AnimalMove* ani = dynamic_cast<AnimalMove*>(check->mpParentObject);
0523: 		if (ani->GetCurrentState() != STATE_VACUUM)
0524: 		{
0525: 			ani->SetTargetPlayer(this);
0526: 			ani->IncreaseVacuumTimer();
0527: 			ani->ChangeStateToVacuum();
0528: 		}
0529: 	}
0530: }

0532: /*
0533:  * @brief コライダー接触終了時のコールバック（吸引範囲外に逃げられた、またはボタンを離した際のクリーンアップ）
0534:  */
0535: void Player3D::OnExit(Collider* collider, Collider* check)
0536: {
0537: 	// --- 牛が範囲外に出た、または吸い込みを中断した場合：地上へ戻す ---
0538: 	if (collider == mpCapsuleCollider && check->mpParentObject->GetTag() == Tag3D_Cow)
0539: 	{
0540: 		CowMove* cow = dynamic_cast<CowMove*>(check->mpParentObject);
0541: 
0542: 		cow->SetTargetPlayer(nullptr);
0543: 		cow->ResetVacuumTimer();            // 蓄積タイマーをリセット
0544: 		cow->SetCurrentState(STATE_WALK);
0545: 		cow->ChangeState(new StateWalk()); // 状態を通常の徘徊歩行に戻す
0546: 
0547: 		VECTOR pos = cow->GetPosition();   // 現在の座標を取得（※必要に応じてY座標を接地させる処理を誘発）
0548: 		cow->SetPosition(pos);
0549: 	}
0550: 	// --- その他動物が範囲外に出た、または吸い込みを中断した場合 ---
0551: 	if (collider == mpCapsuleCollider && check->mpParentObject->GetTag() == Tag3D_Animal)
0552: 	{
0553: 		AnimalMove* ani = dynamic_cast<AnimalMove*>(check->mpParentObject);
0554: 		ani->SetTargetPlayer(nullptr);
0555: 		ani->SetCurrentState(STATE_WALK);
0556: 		ani->ChangeState(new StateWalk());
0557: 		ani->SetPosition(ani->GetPosition());
0558: 	}
0559: }
